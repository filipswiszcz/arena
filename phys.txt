actor struct:
    - global pos (vec2)
    - rigidybody (struct)
    - collider (struct)
    - bool grounded

rigidbody struct:
    - vel, acc (vec2) // do not store acc?
    - ?mass, drag (float)
    - grav (float) // 2.0f = heavy, 1.0f = norm, 0.5f = floaty, 0.0f = no grav
    - rest (float) // 1.0f = perfect bounce, 0.0f = no bounce
    - fric (float) // 0.9f = slipy, 0.5f = norm, 0.1f = sticky
    - ?force (vec2) // accum all forces, which are later used to calc accel

    // acc = F / mass
    // vel += acc * dt
    // pos += vel * dt

collider struct:
    - local pos (vec2)
    - min/max (vec2)
    - dirs (int32_t [4]) // [N, E, S, W]
